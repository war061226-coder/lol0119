import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter

# Load original file
orig_filename = 'LoL_내전_팀배정관리_6 (3).xlsx'
wb = openpyxl.load_workbook(orig_filename)

# 1. Update '설정' Sheet
if '설정' in wb.sheetnames:
    ws_config = wb['설정']
    # Clear existing data in 설정 sheet
    ws_config.delete_rows(1, ws_config.max_row + 1)
else:
    ws_config = wb.create_sheet('설정')

ws_config.views.sheetView[0].showGridLines = True

# Tier Table
ws_config.cell(row=1, column=1, value="티어")
ws_config.cell(row=1, column=2, value="기본 MMR")

tier_data = [
    ("아이언4", 0), ("아이언3", 0), ("아이언2", 0), ("아이언1", 0),
    ("브론즈4", 0), ("브론즈3", 0), ("브론즈2", 0), ("브론즈1", 0),
    ("실버4", 0), ("실버3", 0), ("실버2", 0), ("실버1", 0),
    ("골드4", 1), ("골드3", 2), ("골드2", 3), ("골드1", 4),
    ("플래티넘4", 5), ("플래티넘3", 6), ("플래티넘2", 7), ("플래티넘1", 8),
    ("에메랄드4", 10), ("에메랄드3", 11), ("에메랄드2", 12), ("에메랄드1", 13),
    ("다이아몬드4", 15), ("다이아몬드3", 17), ("다이아몬드2", 19), ("다이아몬드1", 21),
    ("마스터", 25), ("그랜드마스터", 30), ("챌린저", 35)
]

for idx, (t, m) in enumerate(tier_data, start=2):
    ws_config.cell(row=idx, column=1, value=t)
    ws_config.cell(row=idx, column=2, value=m)

# Position Penalty Table
ws_config.cell(row=1, column=4, value="구분")
ws_config.cell(row=1, column=5, value="조정 점수")

adj_data = [
    ("주 포지션", 0),
    ("부 포지션", -2),
    ("타 라인(파괴)", -4)
]

for idx, (g, p) in enumerate(adj_data, start=2):
    ws_config.cell(row=idx, column=4, value=g)
    ws_config.cell(row=idx, column=5, value=p)

# Header Styles for '설정'
header_fill_config = PatternFill(start_color="366092", end_color="366092", fill_type="solid")
header_font_config = Font(name="Calibri", size=11, bold=True, color="FFFFFF")
thin_border = Border(
    left=Side(style='thin', color='D9D9D9'),
    right=Side(style='thin', color='D9D9D9'),
    top=Side(style='thin', color='D9D9D9'),
    bottom=Side(style='thin', color='D9D9D9')
)

for col in [1, 2, 4, 5]:
    cell = ws_config.cell(row=1, column=col)
    cell.fill = header_fill_config
    cell.font = header_font_config
    cell.alignment = Alignment(horizontal="center", vertical="center")

for r in range(1, 33):
    for c in [1, 2]:
        cell = ws_config.cell(row=r, column=c)
        cell.border = thin_border
        if r > 1:
            cell.alignment = Alignment(horizontal="center" if c==1 else "right", vertical="center")

for r in range(1, 5):
    for c in [4, 5]:
        cell = ws_config.cell(row=r, column=c)
        cell.border = thin_border
        if r > 1:
            cell.alignment = Alignment(horizontal="center" if c==4 else "right", vertical="center")

# 2. Add '선수명단(개편)' Sheet
sheet_name = "선수명단(개편)"
if sheet_name in wb.sheetnames:
    del wb[sheet_name]

ws_players = wb.create_sheet(title=sheet_name, index=0) # Place first
ws_players.views.sheetView[0].showGridLines = True

# Title block
ws_players.merge_cells("A1:G1")
title_cell = ws_players.cell(row=1, column=1, value="LoL 내전 선수 명단 및 MMR 자동 계산기")
title_cell.font = Font(name="Calibri", size=15, bold=True, color="1F497D")
title_cell.alignment = Alignment(horizontal="left", vertical="center")

# Headers
headers = ["선수 이름", "티어", "주 포지션", "부 포지션", "출전 포지션", "감점 계산", "최종 MMR"]
header_fill_players = PatternFill(start_color="1F497D", end_color="1F497D", fill_type="solid")
header_font_players = Font(name="Calibri", size=11, bold=True, color="FFFFFF")

for c_idx, h_text in enumerate(headers, start=1):
    cell = ws_players.cell(row=3, column=c_idx, value=h_text)
    cell.fill = header_fill_players
    cell.font = header_font_players
    cell.alignment = Alignment(horizontal="center", vertical="center")

# Sample Data (10 players)
sample_players = [
    ("말라리", "브론즈2", "TOP", "MID", "TOP"),
    ("리코", "골드2", "MID", "ADC", "MID"),
    ("스카이", "에메랄드1", "TOP", "JG", "JG"),     # 부포지션 (-2점)
    ("벌우주", "실버2", "TOP", "MID", "TOP"),
    ("아세", "실버3", "MID", "ADC", "MID"),
    ("김토토", "실버2", "SUP", "없음", "SUP"),
    ("정정쓰", "플래티넘2", "ADC", "JG", "TOP"),    # 타라인 (-4점)
    ("광어", "브론즈3", "SUP", "MID", "SUP"),
    ("참가자9", "골드1", "ADC", "SUP", "ADC"),
    ("참가자10", "다이아몬드4", "JG", "MID", "JG"),
]

for idx, p in enumerate(sample_players, start=4):
    ws_players.cell(row=idx, column=1, value=p[0])
    ws_players.cell(row=idx, column=2, value=p[1])
    ws_players.cell(row=idx, column=3, value=p[2])
    ws_players.cell(row=idx, column=4, value=p[3])
    ws_players.cell(row=idx, column=5, value=p[4])
    
    # F열 (감점 계산 수식)
    f_formula = f'=IF(E{idx}="","", IF(E{idx}=C{idx}, 설정!$E$2, IF(E{idx}=D{idx}, 설정!$E$3, 설정!$E$4)))'
    ws_players.cell(row=idx, column=6, value=f_formula)
    
    # G열 (최종 MMR 수식)
    g_formula = f'=IF(B{idx}="","", MAX(0, XLOOKUP(B{idx}, 설정!$A$2:$A$32, 설정!$B$2:$B$32, 0) + F{idx}))'
    ws_players.cell(row=idx, column=7, value=g_formula)

for r in range(4, 14):
    for c in range(1, 8):
        cell = ws_players.cell(row=r, column=c)
        cell.border = thin_border
        cell.alignment = Alignment(horizontal="center", vertical="center")
        cell.font = Font(name="Calibri", size=11)

# Team Totals & Difference
ws_players.cell(row=15, column=5, value="A팀 MMR 합계 (1~5번):").alignment = Alignment(horizontal="right", vertical="center")
ws_players.cell(row=15, column=5).font = Font(name="Calibri", size=10, bold=True)
ws_players.cell(row=15, column=7, value="=SUM(G4:G8)").font = Font(name="Calibri", size=11, bold=True)
ws_players.cell(row=15, column=7).alignment = Alignment(horizontal="center", vertical="center")
ws_players.cell(row=15, column=7).fill = PatternFill(start_color="DCE6F1", end_color="DCE6F1", fill_type="solid")
ws_players.cell(row=15, column=7).border = thin_border

ws_players.cell(row=16, column=5, value="B팀 MMR 합계 (6~10번):").alignment = Alignment(horizontal="right", vertical="center")
ws_players.cell(row=16, column=5).font = Font(name="Calibri", size=10, bold=True)
ws_players.cell(row=16, column=7, value="=SUM(G9:G13)").font = Font(name="Calibri", size=11, bold=True)
ws_players.cell(row=16, column=7).alignment = Alignment(horizontal="center", vertical="center")
ws_players.cell(row=16, column=7).fill = PatternFill(start_color="F2DCDB", end_color="F2DCDB", fill_type="solid")
ws_players.cell(row=16, column=7).border = thin_border

ws_players.cell(row=17, column=5, value="양 팀 MMR 차이 (밸런스):").alignment = Alignment(horizontal="right", vertical="center")
ws_players.cell(row=17, column=5).font = Font(name="Calibri", size=10, bold=True, color="C00000")
ws_players.cell(row=17, column=7, value="=ABS(G15-G16)").font = Font(name="Calibri", size=11, bold=True, color="C00000")
ws_players.cell(row=17, column=7).alignment = Alignment(horizontal="center", vertical="center")
ws_players.cell(row=17, column=7).fill = PatternFill(start_color="FCE4D6", end_color="FCE4D6", fill_type="solid")
ws_players.cell(row=17, column=7).border = thin_border

# Adjust Column Widths
for ws in [ws_players, ws_config]:
    for col in ws.columns:
        max_len = 0
        col_letter = get_column_letter(col[0].column)
        for cell in col:
            val_str = str(cell.value or '')
            if cell.row == 1 and ws == ws_players:
                continue
            length = sum(2 if ord(char) > 127 else 1 for char in val_str)
            if length > max_len:
                max_len = length
        ws.column_dimensions[col_letter].width = max(max_len + 4, 12)

ws_players.column_dimensions['E'].width = 16
ws_players.column_dimensions['F'].width = 14
ws_players.column_dimensions['G'].width = 14

# Save merged workbook
merged_filename = 'LoL_내전_팀배정관리_개편완료.xlsx'
wb.save(merged_filename)
print(f"Successfully merged into {merged_filename}")
